### 2.2 核心服务职责

#### InputMonitorService（新增）
- 管理Windows低级别输入钩子
- 在后台线程运行消息循环
- 实时统计按键/鼠标数据
- 触发活跃度状态变化事件

#### UsageStatisticsService（新增）
- 整合输入统计数据
- 计算活跃/空闲时长
- 提供查询接口
- 数据持久化

namespace eyesharp.Models
{
    /// <summary>
    /// 输入事件原始数据（内存中实时累计，不序列化）
    /// </summary>
    public class InputEventCounter
    {
        public int KeyPressCount { get; set; }
        public long MouseMoveDistance { get; set; }
        public int MouseLeftClickCount { get; set; }
        public int MouseRightClickCount { get; set; }
        public int MouseMiddleClickCount { get; set; }
        public int MouseWheelScrollCount { get; set; }

        // 上次鼠标位置，用于计算移动距离
        public System.Drawing.Point LastMousePosition { get; set; }
        public bool HasLastPosition { get; set; }

        /// <summary>
        /// 重置计数器
        /// </summary>
        public void Reset()
        {
            KeyPressCount = 0;
            MouseMoveDistance = 0;
            MouseLeftClickCount = 0;
            MouseRightClickCount = 0;
            MouseMiddleClickCount = 0;
            MouseWheelScrollCount = 0;
            HasLastPosition = false;
        }

        /// <summary>
        /// 累加另一个计数器的数据
        /// </summary>
        public void Add(InputEventCounter other)
        {
            KeyPressCount += other.KeyPressCount;
            MouseMoveDistance += other.MouseMoveDistance;
            MouseLeftClickCount += other.MouseLeftClickCount;
            MouseRightClickCount += other.MouseRightClickCount;
            MouseMiddleClickCount += other.MouseMiddleClickCount;
            MouseWheelScrollCount += other.MouseWheelScrollCount;
        }
    }

    /// <summary>
    /// 小时级详细记录（持久化存储）
    /// </summary>
    public class HourlyActivityRecord
    {
        public DateTime Hour { get; set; }

        // 时间分布
        public int ActiveSeconds { get; set; }
        public int IdleSeconds { get; set; }
        public int LockScreenSeconds { get; set; }
        public int LockScreenCount { get; set; }

        // 输入统计
        public int KeyPressCount { get; set; }
        public long MouseMoveDistance { get; set; }
        public int MouseLeftClickCount { get; set; }
        public int MouseRightClickCount { get; set; }
        public int MouseMiddleClickCount { get; set; }
        public int MouseWheelScrollCount { get; set; }

        // 状态标记
        public bool IsCompleteHour { get; set; } = true;  // 是否完整小时数据
        public DateTime LastUpdateTime { get; set; }      // 最后更新时间
    }

    /// <summary>
    /// 日汇总统计（持久化存储）
    /// </summary>
    public class DailyUsageStatistics
    {
        public DateTime Date { get; set; }

        // 时间统计
        public int TotalBootSeconds { get; set; }
        public int ActiveSeconds { get; set; }
        public int IdleSeconds { get; set; }
        public int LockScreenSeconds { get; set; }
        public int LockScreenCount { get; set; }

        // 输入统计汇总
        public int TotalKeyPressCount { get; set; }
        public long TotalMouseMoveDistance { get; set; }
        public int TotalMouseLeftClickCount { get; set; }
        public int TotalMouseRightClickCount { get; set; }
        public int TotalMouseMiddleClickCount { get; set; }
        public int TotalMouseWheelScrollCount { get; set; }

        // 24小时详细数据（仅保留最近90天）
        public List<HourlyActivityRecord>? HourlyData { get; set; }

        // 元数据
        public DateTime FirstRecordTime { get; set; }
        public DateTime LastRecordTime { get; set; }
        public int Version { get; set; } = 1;
    }

    /// <summary>
    /// 实时状态（内存中，用于UI显示）
    /// </summary>
    public class RealTimeUsageStatus
    {
        public DateTime CurrentHour { get; set; }
        public UserActivityState CurrentState { get; set; }
        public DateTime StateStartTime { get; set; }

        // 今日累计（从内存计算）
        public TimeSpan TodayActiveTime { get; set; }
        public TimeSpan TodayIdleTime { get; set; }
        public InputEventCounter TodayInputCounter { get; set; } = new();

        // 当前状态持续时间
        public TimeSpan CurrentStateDuration => DateTime.Now - StateStartTime;

-- 说明: EyeSharp使用统计数据库

-- 数据库元数据表
CREATE TABLE IF NOT EXISTS DbMetadata (
    Key TEXT PRIMARY KEY,

-- 插入版本信息
INSERT OR REPLACE INTO DbMetadata (Key, Value) VALUES
    ('Version', '1'),
    ('CreatedAt', datetime('now')),
    ('MachineName', @MachineName),
    ('UserName', @UserName);

-- 表: HourlyRecords (小时级详细记录)
-- 说明: 每小时的详细活动数据
-- 保留: 90天
-- =============================================
CREATE TABLE IF NOT EXISTS HourlyRecords (
    Id INTEGER PRIMARY KEY AUTOINCREMENT,
    Hour DATETIME NOT NULL UNIQUE,              -- 小时时间段 (如 2026-03-08 14:00)

    -- 时间分布
    ActiveSeconds INTEGER DEFAULT 0,            -- 活跃秒数
    IdleSeconds INTEGER DEFAULT 0,              -- 空闲秒数
    LockScreenSeconds INTEGER DEFAULT 0,        -- 锁屏秒数
    LockScreenCount INTEGER DEFAULT 0,          -- 锁屏次数

    -- 输入统计
    KeyPressCount INTEGER DEFAULT 0,            -- 键盘按键次数
    MouseMoveDistance INTEGER DEFAULT 0,        -- 鼠标移动距离（像素）
    MouseLeftClickCount INTEGER DEFAULT 0,      -- 左键点击
    MouseRightClickCount INTEGER DEFAULT 0,     -- 右键点击
    MouseMiddleClickCount INTEGER DEFAULT 0,    -- 中键点击
    MouseWheelScrollCount INTEGER DEFAULT 0,    -- 滚轮滚动

    -- 元数据
    IsCompleteHour BOOLEAN DEFAULT 1,           -- 是否完整小时
    CreatedAt DATETIME DEFAULT CURRENT_TIMESTAMP,
    UpdatedAt DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 索引: 按时间查询
CREATE INDEX IF NOT EXISTS idx_hourly_hour ON HourlyRecords(Hour);
CREATE INDEX IF NOT EXISTS idx_hourly_date ON HourlyRecords(date(Hour));

-- =============================================
-- 表: DailySummaries (日级汇总)
-- 说明: 每日数据汇总，长期保留
-- 保留: 365天
-- =============================================
CREATE TABLE IF NOT EXISTS DailySummaries (
    Id INTEGER PRIMARY KEY AUTOINCREMENT,
    Date DATE NOT NULL UNIQUE,                  -- 日期

    -- 时间统计
    TotalBootSeconds INTEGER DEFAULT 0,         -- 总开机时长
    ActiveSeconds INTEGER DEFAULT 0,            -- 活跃时长
    IdleSeconds INTEGER DEFAULT 0,              -- 空闲时长
    LockScreenSeconds INTEGER DEFAULT 0,        -- 锁屏总时长
    LockScreenCount INTEGER DEFAULT 0,          -- 锁屏次数

    -- 输入统计汇总
    TotalKeyPressCount INTEGER DEFAULT 0,
    TotalMouseMoveDistance INTEGER DEFAULT 0,
    TotalMouseLeftClickCount INTEGER DEFAULT 0,
    TotalMouseRightClickCount INTEGER DEFAULT 0,
    TotalMouseMiddleClickCount INTEGER DEFAULT 0,
    TotalMouseWheelScrollCount INTEGER DEFAULT 0,

    -- 元数据
    FirstRecordTime DATETIME,                   -- 首次记录时间
    LastRecordTime DATETIME,                    -- 最后记录时间
    CreatedAt DATETIME DEFAULT CURRENT_TIMESTAMP,
    UpdatedAt DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 索引: 按日期查询
CREATE INDEX IF NOT EXISTS idx_daily_date ON DailySummaries(Date);

-- =============================================
-- 视图: 最近30天汇总（用于快速展示）
-- =============================================
CREATE VIEW IF NOT EXISTS Recent30DaysSummary AS
SELECT
    Date,
    printf('%02d:%02d', TotalBootSeconds / 3600, (TotalBootSeconds % 3600) / 60) as BootTime,
    printf('%02d:%02d', ActiveSeconds / 3600, (ActiveSeconds % 3600) / 60) as ActiveTime,
    TotalKeyPressCount,
    printf('%.2f', TotalMouseMoveDistance / 3779.5) as MouseMoveMeters  -- 转换为米
FROM DailySummaries
WHERE Date >= date('now', '-30 days')
ORDER BY Date DESC;

-- =============================================
-- 清理存储过程（每小时执行）
-- =============================================
-- 删除90天前的详细小时数据
DELETE FROM HourlyRecords
WHERE Hour < datetime('now', '-90 days');

-- 删除365天前的日汇总
DELETE FROM DailySummaries
WHERE Date < date('now', '-365 days');

### 3.3 数据库访问层设计

```csharp
namespace eyesharp.Services
{
    /// <summary>
    /// SQLite数据库访问上下文
    /// </summary>
    public class UsageStatisticsDbContext : IDisposable
    {
        private readonly SqliteConnection _connection;
        private readonly ILogService _logService;
        private readonly string _dbPath;

        public UsageStatisticsDbContext(ILogService logService)
        {
            _logService = logService;
            var baseDir = AppDomain.CurrentDomain.BaseDirectory;
            _dbPath = Path.Combine(baseDir, "usage_statistics.db");
            _connection = new SqliteConnection($"Data Source={_dbPath}");
        }

        /// <summary>
        /// 初始化数据库（创建表和索引）
        /// </summary>
        public async Task InitializeAsync()
        {
            await _connection.OpenAsync();

            using var cmd = _connection.CreateCommand();
            cmd.CommandText = SqlScripts.CreateTables;
            await cmd.ExecuteNonQueryAsync();

            _logService.Info($"[UsageStatsDb] 数据库初始化完成: {_dbPath}");
        }

        /// <summary>
        /// 保存或更新小时记录
        /// </summary>
        public async Task SaveHourlyRecordAsync(HourlyActivityRecord record)
        {
            using var cmd = _connection.CreateCommand();
            cmd.CommandText = @"
                INSERT INTO HourlyRecords (
                    Hour, ActiveSeconds, IdleSeconds, LockScreenSeconds, LockScreenCount,
                    KeyPressCount, MouseMoveDistance, MouseLeftClickCount, MouseRightClickCount,
                    MouseMiddleClickCount, MouseWheelScrollCount, IsCompleteHour, UpdatedAt
                ) VALUES (
                    @Hour, @ActiveSeconds, @IdleSeconds, @LockScreenSeconds, @LockScreenCount,
                    @KeyPressCount, @MouseMoveDistance, @MouseLeftClickCount, @MouseRightClickCount,
                    @MouseMiddleClickCount, @MouseWheelScrollCount, @IsCompleteHour, @UpdatedAt
                )
                ON CONFLICT(Hour) DO UPDATE SET
                    ActiveSeconds = ActiveSeconds + excluded.ActiveSeconds,
                    IdleSeconds = IdleSeconds + excluded.IdleSeconds,
                    LockScreenSeconds = LockScreenSeconds + excluded.LockScreenSeconds,
                    LockScreenCount = LockScreenCount + excluded.LockScreenCount,
                    KeyPressCount = KeyPressCount + excluded.KeyPressCount,
                    MouseMoveDistance = MouseMoveDistance + excluded.MouseMoveDistance,
                    MouseLeftClickCount = MouseLeftClickCount + excluded.MouseLeftClickCount,
                    MouseRightClickCount = MouseRightClickCount + excluded.MouseRightClickCount,
                    MouseMiddleClickCount = MouseMiddleClickCount + excluded.MouseMiddleClickCount,
                    MouseWheelScrollCount = MouseWheelScrollCount + excluded.MouseWheelScrollCount,
                    UpdatedAt = excluded.UpdatedAt;
            ";

            // 添加参数...
            await cmd.ExecuteNonQueryAsync();
        }

        /// <summary>
        /// 获取指定日期范围的小时记录
        /// </summary>
        public async Task<List<HourlyActivityRecord>> GetHourlyRecordsAsync(
            DateTime startDate, DateTime endDate)
        {
            using var cmd = _connection.CreateCommand();
            cmd.CommandText = @"
                SELECT * FROM HourlyRecords
                WHERE Hour >= @StartDate AND Hour <= @EndDate
                ORDER BY Hour;
            ";
            cmd.Parameters.AddWithValue("@StartDate", startDate);
            cmd.Parameters.AddWithValue("@EndDate", endDate);

            var results = new List<HourlyActivityRecord>();
            using var reader = await cmd.ExecuteReaderAsync();
            while (await reader.ReadAsync())
            {
                results.Add(MapHourlyRecord(reader));
            }
            return results;
        }

        /// <summary>
        /// 清理过期数据
        /// </summary>
        public async Task CleanupOldDataAsync()
        {
            using var transaction = _connection.BeginTransaction();
            try
            {
                // 删除90天前的详细数据
                using (var cmd = _connection.CreateCommand())
                {
                    cmd.CommandText = "DELETE FROM HourlyRecords WHERE Hour < datetime('now', '-90 days');";
                    var hourlyDeleted = await cmd.ExecuteNonQueryAsync();
                    _logService.Debug($"[UsageStatsDb] 清理了{hourlyDeleted}条过期小时记录");
                }

                // 删除365天前的日汇总
                using (var cmd = _connection.CreateCommand())
                {
                    cmd.CommandText = "DELETE FROM DailySummaries WHERE Date < date('now', '-365 days');";
                    var dailyDeleted = await cmd.ExecuteNonQueryAsync();
                    _logService.Debug($"[UsageStatsDb] 清理了{dailyDeleted}条过期日记录");
                }

                // 执行VACUUM释放空间
                using (var cmd = _connection.CreateCommand())
                {
                    cmd.CommandText = "VACUUM;";
                    await cmd.ExecuteNonQueryAsync();
                }

                transaction.Commit();
            }
            catch (Exception ex)
            {
                transaction.Rollback();
                _logService.Error(ex, "[UsageStatsDb] 清理过期数据失败");
                throw;
            }
        }

        public void Dispose()
        {
            _connection?.Dispose();
        }
    }
}

### 3.4 数据生命周期管理

```
当前小时计数器（内存）
        │
        ▼ 每小时整点/程序退出时
HourlyRecords表（SQLite）
        │
        ▼ 每日24点触发
DailySummaries表（SQLite）
        │
    ┌───┴───┐
    │       │
    ▼       ▼
保留90天  保留365天
详细数据  汇总数据
    │       │
    ▼       ▼
自动清理  自动清理
（SQL删除）（SQL删除）
```

**SQLite数据管理策略：**

1. **写入策略**：使用UPSERT（`INSERT ... ON CONFLICT ... DO UPDATE`）实现幂等写入
2. **事务控制**：批量操作使用事务保证一致性
3. **清理策略**：定时执行SQL删除过期数据，随后执行`VACUUM`释放空间
4. **索引优化**：时间字段建立索引，支持快速范围查询
5. **WAL模式**：启用Write-Ahead Logging提升并发性能

```sql
-- 启用WAL模式（提升写入性能）
PRAGMA journal_mode = WAL;
PRAGMA synchronous = NORMAL;
```

## 五、异常处理设计

### 5.1 输入监控异常

| 异常场景 | 处理方式 | 日志级别 |
|----------|----------|----------|
| 钩子安装失败（权限不足） | 降级到定时检测模式；提示用户添加白名单 | Error |
| 钩子被其他程序卸载 | 检测到后自动重连；最多重试3次 | Warning |
| 消息泵线程崩溃 | 自动重启线程；保留当前计数 | Error |
| 内存溢出风险 | 超过阈值时强制保存并重置计数 | Warning |

### 5.2 数据库异常

| 异常场景 | 处理方式 | 日志级别 |
|----------|----------|----------|
| 数据库锁定（并发写入） | SQLite自动处理；事务重试 | Warning |
| 磁盘空间不足 | 清理过期数据；仍不足则提示用户 | Error |
| 数据库损坏 | 备份损坏文件；重新创建新数据库 | Error |
| SQL执行错误 | 记录详细SQL和参数；跳过当前操作 | Error |

```csharp
// 数据库损坏恢复
public async Task RecoverFromCorruptionAsync() {
    try {
        // 备份损坏的数据库
        var backupPath = $"{_dbPath}.corrupted_{DateTime.Now:yyyyMMddHHmmss}";
        File.Move(_dbPath, backupPath);
        _logService.Warning($"[UsageStatsDb] 已备份损坏的数据库: {backupPath}");

        // 重新初始化
        await InitializeAsync();
        _logService.Info("[UsageStatsDb] 数据库已重建");
    } catch (Exception ex) {
        _logService.Error(ex, "[UsageStatsDb] 数据库恢复失败");
    }
}

// 事务重试机制
public async Task ExecuteWithRetryAsync(Func<Task> action, int maxRetries = 3) {
    for (int i = 0; i < maxRetries; i++) {
        try {
            await action();
            return;
        } catch (SqliteException ex) when (ex.SqliteErrorCode == 5) { // SQLITE_BUSY
            _logService.Warning($"[UsageStatsDb] 数据库忙，{i+1}/{maxRetries}次重试...");
            await Task.Delay(100 * (i + 1)); // 指数退避
        }
    }
    throw new Exception("达到最大重试次数");
}

## 八、关键设计决策

### 8.1 为什么使用低级别钩子而非轮询？

| 方案 | 优点 | 缺点 |
|------|------|------|
| **低级别钩子** | 实时性好，精确度高 | 可能被安全软件拦截 |
| **定时轮询** | 实现简单，兼容性好 | 精度差（5秒间隔），无法统计移动距离 |

**决策**：优先使用低级别钩子，失败时降级到轮询模式。

### 8.2 为什么用独立线程而非UI线程？

- Windows钩子需要在消息循环中处理回调
- UI线程可能被阻塞，影响响应
- 独立线程可保证数据收集的连续性

### SQLCipher密钥管理方案

#### 方案一：机器特征派生密钥（推荐）
```csharp
public class DatabaseEncryptionKeyProvider
{
    /// <summary>
    /// 从机器特征派生加密密钥
    /// 优点：无需用户设置密码，密钥与机器绑定
    /// 缺点：数据库文件无法迁移到其他机器
    /// </summary>
    public static string GetEncryptionKey()
    {
        // 组合多个机器特征
        var cpuId = GetCpuId();           // CPU ID
        var diskSerial = GetDiskSerial(); // 系统盘序列号
        var machineGuid = GetMachineGuid(); // Windows MachineGuid

        // 组合并哈希
        var combined = $"{cpuId}:{diskSerial}:{machineGuid}:EyeSharpSalt_v1";

        using var sha256 = SHA256.Create();
        var hash = sha256.ComputeHash(Encoding.UTF8.GetBytes(combined));
        return Convert.ToHexString(hash);
    }
}
```

#### 方案二：用户设置密码
```csharp
public class DatabaseEncryptionKeyProvider
{
    /// <summary>
    /// 从用户密码派生密钥
    /// 优点：用户可迁移数据文件到任意机器
    /// 缺点：需要用户记住密码，首次使用需设置
    /// </summary>
    public static string DeriveKeyFromPassword(string password)
    {
        // 使用PBKDF2派生密钥
        using var pbkdf2 = new Rfc2898DeriveBytes(
            password,
            Encoding.UTF8.GetBytes("EyeSharpFixedSalt_v1"), // 固定盐值
            iterations: 100000,
            HashAlgorithmName.SHA256
        );
        var key = pbkdf2.GetBytes(32); // 256位密钥
        return Convert.ToHexString(key);
    }
}
```

**当前选择**: 方案一（机器特征派生）
- 用户无感知
- 数据与机器绑定（符合"本地使用"场景）
- 密钥不存储，运行时派生

```csharp
/// <summary>
/// 更改数据库密码（数据导出再导入）
/// </summary>
public async Task ChangeEncryptionKeyAsync(string newKey)
{
    var backupPath = _dbPath + ".backup";

    // 1. 导出当前数据
    await ExportToSqlAsync(backupPath);

    // 2. 关闭当前连接
    _connection.Close();

    // 3. 删除原数据库
    File.Delete(_dbPath);

    // 4. 使用新密钥创建新数据库
    var newConnection = new SqliteConnection(
        $"Data Source={_dbPath};Password={newKey};"
    );

    // 5. 导入数据
    await ImportFromSqlAsync(newConnection, backupPath);

    // 6. 清理备份
    File.Delete(backupPath);
}
